// ── trace facts ──────────────────────────────────────────────────────────────
interface TraceFacts { files: number; llmRequests: number; numCtx: number[]; iterations: number; surfaces: string[]; endStatus: string | null; toolCalls: number; flags: Record<string, unknown>[]; guards: Record<string, number>; reconFindings: number; catalogLine: boolean }
function traceFacts(dir: string): TraceFacts {
  const f: TraceFacts = { files: 0, llmRequests: 0, numCtx: [], iterations: 0, surfaces: [], endStatus: null, toolCalls: 0, flags: [], guards: {}, reconFindings: 0, catalogLine: false }
  if (!existsSync(dir)) return f
  for (const name of readdirSync(dir).filter((n) => n.endsWith(".jsonl")).sort()) {
    f.files++
    const text = readFileSync(join(dir, name), "utf8")
    if (/tool: name/.test(text)) f.catalogLine = true
    for (const line of text.split("\n")) {
      if (!line.trim()) continue
      let e: Record<string, unknown>; try { e = JSON.parse(line) as Record<string, unknown> } catch { continue }
      const k = e.kind
      if (k === "guard-fired") {
        const g = String(e.guard); f.guards[g] = (f.guards[g] ?? 0) + 1
        if (g === "wire-request") { f.llmRequests++; const n = ((e.metadata as { numCtx?: number } | undefined) ?? {}).numCtx; if (typeof n === "number" && !f.numCtx.includes(n)) f.numCtx.push(n) }
      } else if (k === "tool-surface-resolved") { f.iterations++; const v = (e.visible ?? e.tools ?? e.names) as unknown; if (Array.isArray(v) && f.surfaces.length < 6) f.surfaces.push(v.map(String).join("+")) }
      else if (k === "run-completed") f.endStatus = String(e.status ?? "")
      else if (k === "run-started") { const c = (e.config ?? {}) as { mechanisms?: Record<string, unknown>; posture?: unknown }; f.flags.push({ posture: c.posture ?? "execute", ...Object.fromEntries(["planToolCatalog", "planBudget", "planRecon", "wrapUpReserve", "leanThinking", "budgetAnswerTurn"].map((x) => [x, (c.mechanisms ?? {})[x]])) }) }
      else if (k === "tool-call-start") f.toolCalls++
      if (k === "ledger-entry" && /plan-finding/.test(line)) f.reconFindings++
    }
  }
  return f
}

// ── one cell ─────────────────────────────────────────────────────────────────
// eslint-disable-next-line @typescript-eslint/no-explicit-any
const mod = await import(`${root}/packages/runtime/src/index.ts`) as { ReactiveAgents: { create(): any } }
async function runCell(task: Task, run: number, cellDir: string): Promise<Record<string, unknown>> {
  const work = join(WORK, `${process.pid}-${task.id}-${run}`)
  rmSync(work, { recursive: true, force: true }); mkdirSync(work, { recursive: true })
  for (const [p, c] of Object.entries(task.fixtures)) { mkdirSync(dirname(join(work, p)), { recursive: true }); writeFileSync(join(work, p), c) }
  rmSync(cellDir, { recursive: true, force: true }); mkdirSync(cellDir, { recursive: true })
  const t0 = performance.now()
  let billed = 0, llmCalls = 0, sawLLM = false, planMs = 0, planSteps = -1, planErr: string | undefined, execErr: string | undefined, output = "", success: boolean | undefined
  const strat = mode.startsWith("strategy:") ? mode.slice(9) : "reactive"
  const builder = mod.ReactiveAgents.create().withName(`pp-${task.id}`).withProvider("ollama").withModel(model).withMaxIterations(task.maxIterations)
    .withTools({ builtins: ["file-read", "file-write"], workspace: work }).withReasoning({ defaultStrategy: strat, enableStrategySwitching: false })
    .withTracing({ dir: cellDir }).withKillSwitch().withTimeout(Math.round(cellTimeoutMs / 2))
  const _log = console.log; console.log = () => {}
  const agent = await builder.build(); console.log = _log
  const unsub = await agent.subscribe((ev: { _tag: string; billedTokens?: number; tokensUsed?: number }) => { if (ev._tag === "LLMRequestCompleted") { billed += ev.billedTokens ?? ev.tokensUsed ?? 0; llmCalls++; sawLLM = true } })
  const prompt = `Working directory for this task: ${work}\n\n${task.prompt(work)}`
  const body = async (): Promise<void> => {
    if (mode === "plan") {
      const p0 = performance.now()
      let plan: { steps: unknown[] } | undefined
      try { plan = await agent.plan(prompt); planSteps = plan!.steps.length } catch (e) { planErr = e instanceof Error ? `${e.name}:${e.message}`.slice(0, 160) : String(e) }
      planMs = performance.now() - p0
      if (plan) { try { const r = await agent.execute(plan); output = r.output; success = r.success } catch (e) { execErr = e instanceof Error ? e.message.slice(0, 160) : String(e) } }
    } else {
      try { const r = await agent.run(prompt); output = r.output; success = r.success } catch (e) { execErr = e instanceof Error ? e.message.slice(0, 160) : String(e) }
    }
  }
  let timedOut = false
  let timer: ReturnType<typeof setTimeout> | undefined
  await Promise.race([body(), new Promise<void>((res) => { timer = setTimeout(() => { timedOut = true; res() }, cellTimeoutMs) })])
  clearTimeout(timer)
  if (timedOut) { try { await agent.terminate?.() } catch { /* no-op */ } }
  const durationMs = performance.now() - t0
  unsub(); try { await agent.dispose() } catch { /* no-op */ }
  const g = task.grade(work).checks
  const tf = traceFacts(cellDir)
  rmSync(work, { recursive: true, force: true })
  const passN = g.filter(([, ok]) => ok).length
  return {
    taskId: task.id, run, accuracy: passN === g.length ? 1 : 0, fraction: passN / g.length, failedChecks: g.filter(([, ok]) => !ok).map(([n]) => n),
    billedTokens: sawLLM ? billed : null, llmCalls, durationMs: Math.round(durationMs), planMs: Math.round(planMs), planSteps, planErr, execErr, timedOut, success,
    outputHead: output.slice(0, 120), trace: tf,
    inconclusive: tf.files === 0 || llmCalls === 0,
  }
}

// ── main ─────────────────────────────────────────────────────────────────────
interface Results { meta: Record<string, unknown>; cells: Record<string, unknown>[] }
const res: Results = existsSync(out) ? JSON.parse(readFileSync(out, "utf8")) : { meta: { arm: armName, mode, model, root, runs, startedAt: new Date().toISOString(), env: ARMS[armName] }, cells: [] }
const persist = (): void => { mkdirSync(dirname(out), { recursive: true }); writeFileSync(`${out}.tmp`, JSON.stringify(res, null, 2)); renameSync(`${out}.tmp`, out) }
const tasks = TASKS.filter((t) => !taskFilter || taskFilter.includes(t.id))
const start = Date.now(); let ran = 0
outer: for (let run = 0; run < runs; run++) for (const t of tasks) {
  const id = `${armName}|${mode}|${model}|${t.id}|${run}`
  if (res.cells.some((c) => c.cellId === id && !c.voided)) continue
  if ((Date.now() - start) / 1000 + cellTimeoutMs / 1000 * 0.5 > budgetSec) break outer
  const cell = await runCell(t, run, join(traceRoot, `${t.id}-r${run}`))
  res.cells.push({ cellId: id, arm: armName, mode, model, root, ...cell }); persist(); ran++
  console.error(`${id} acc=${cell.accuracy} frac=${(cell.fraction as number).toFixed(2)} calls=${cell.llmCalls} tok=${cell.billedTokens} ${Math.round((cell.durationMs as number) / 1000)}s steps=${cell.planSteps}${cell.timedOut ? " TIMEOUT" : ""}${cell.planErr ? " PLANERR" : ""}${cell.execErr ? " EXECERR" : ""}`)
  if (arg("pilot")) break outer
}
console.error(JSON.stringify({ ran, remaining: tasks.length * runs - res.cells.filter((c) => !c.voided).length }))
process.exit(0)
