import analyze
_orig = analyze.load
MAXRUN = [99]; TASKS = [None]
def _l(tag, model):
    cs = _orig(tag, model)
    return None if cs is None else [c for c in cs if (TASKS[0] is None or c["taskId"] in TASKS[0]) and c["run"] < MAXRUN[0]]
analyze.load = _l
def trig_rate(tag, model, guard, maxrun=99):
    cs = [c for c in (_orig(tag, model) or []) if not c.get("inconclusive") and c["run"] < maxrun]
    if not cs: return None
    return round(sum(1 for c in cs if any(guard in g for g in c["trace"]["guards"])) / len(cs), 2), len(cs)
def line(ref, cand, model, tasks=None, label="", maxrun=99):
    MAXRUN[0] = maxrun; TASKS[0] = tasks
    d = analyze.compare(ref, cand, model)
    if "missing" in d: return f"{label} {model} MISSING {d['missing']}"
    return (f"{label:32s} {model:13s} n={d['nA']}/{d['nB']} acc {d['accA']:.2f}->{d['accB']:.2f} lift {d['lift_pp']:+.1f}pp sigma {d['sigma_pp']} z={d['z']} | calls {d['callsA']}->{d['callsB']} ({d['calls_change_pct']:+}%, paired {d['calls_paired_diff']} ci95 {d['calls_diff_ci95']}) | billed tok {d['tokA']}->{d['tokB']} ({d['tok_change_pct']:+}%) | timeouts {d['timeoutsA']}/{d['timeoutsB']} voided {d['voidedA']}/{d['voidedB']}")
def accs(tag, model, maxrun=99, tasks=None):
    cs = [c for c in (_orig(tag, model) or []) if not c.get("inconclusive") and c["run"] < maxrun and (tasks is None or c["taskId"] in tasks)]
    return f"{sum(c['accuracy'] for c in cs)}/{len(cs)}"
