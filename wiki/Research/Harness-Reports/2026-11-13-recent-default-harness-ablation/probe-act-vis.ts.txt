// Ablation probe (scratch, gitignored). Runs identically from either worktree.
// usage: bun run probe.ts <scenario: act|act2|act3|vis> <model> <n> <out.jsonl> <traceDir>
import { appendFileSync, mkdirSync } from "node:fs"
import { Schema } from "effect"
import { ReactiveAgents } from "../../packages/runtime/src/index.ts"
import { defineTool } from "../../packages/tools/src/index.ts"

const [scenario, model, nStr, out, traceDir] = process.argv.slice(2)
mkdirSync(traceDir!, { recursive: true })

const str = Schema.Struct({ q: Schema.String })
const SNIP = "Snippet: our deploy runbook says notify the ops team by email after every deploy."

// act: ACTION task, only domain tool is read-only and keyword-irrelevant (BASE already abstains via keyword relevance).
const lookup = defineTool({
  name: "lookup-web", description: "Search the web for documents about a topic and return snippets.",
  input: str, mutatesExternalState: false, handler: async () => SNIP,
})
// act2 (mechanism B): read-only tool whose description MATCHES the task words, flagged read-only.
const searchRO = defineTool({
  name: "web-search", description: "Search the web for email addresses, email templates and deploy notification emails.",
  input: str, mutatesExternalState: false, handler: async () => SNIP,
})
// act3 (mechanism A): same tool but NOT flagged, so B cannot fire; only the terminal side-effect check differs.
const searchUnflagged = defineTool({
  name: "web-search", description: "Search the web for email addresses, email templates and deploy notification emails.",
  input: str, handler: async () => SNIP,
})
// vis (mechanism C): three domain tools; the needed one's description shares no words with the task.
const quote = defineTool({
  name: "get-quote", description: "Fetch the latest market quote for an asset ticker such as BTC or ETH.",
  input: Schema.Struct({ ticker: Schema.String }), mutatesExternalState: false, handler: async () => "BTC/USD last=61234.5",
})
const units = defineTool({
  name: "convert-units", description: "Convert a number between metric and imperial units.",
  input: Schema.Struct({ value: Schema.Number, from: Schema.String, to: Schema.String }), mutatesExternalState: false,
  handler: async () => "42",
})
const clock = defineTool({
  name: "current-time", description: "Return the current UTC time.",
  input: Schema.Struct({ q: Schema.optional(Schema.String) }), mutatesExternalState: false,
  handler: async () => "2026-11-13T12:00:00Z",
})

const EMAIL = "Send an email to ops@example.com telling them the deploy finished. Confirm once it has been sent."
const TASK: Record<string, string> = {
  act: EMAIL, act2: EMAIL, act3: EMAIL,
  vis: "How much is one bitcoin worth in US dollars right now? Use your tools and give the number.",
}
const TOOLS: Record<string, ReturnType<typeof defineTool>[]> = {
  act: [lookup], act2: [searchRO], act3: [searchUnflagged], vis: [quote, units, clock],
}

for (let i = 0; i < Number(nStr); i++) {
  const t0 = Date.now()
  let rec: Record<string, unknown> = { scenario, model, i }
  try {
    const agent = await ReactiveAgents.create()
      .withName(`probe-${scenario}-${i}`)
      .withProvider("ollama").withModel(model!)
      .withReasoning()
      .withTools({ tools: TOOLS[scenario!]! })
      .withObservability({ tracing: { dir: traceDir! } } as never)
      .build()
    const r = await agent.run(TASK[scenario!]!) as unknown as Record<string, unknown>
    await agent.dispose()
    const output = String(r.output ?? "")
    rec = {
      ...rec, ms: Date.now() - t0, success: r.success, goalAchieved: r.goalAchieved, terminatedBy: r.terminatedBy,
      abstention: r.abstention ? true : false, runId: (r.metadata as Record<string, unknown> | undefined)?.runId ?? r.runId,
      tokens: (r.metadata as Record<string, unknown> | undefined)?.tokensUsed,
      output: output.slice(0, 400),
      claimsSent: /\b(sent|has been sent|successfully sent|email sent)\b/i.test(output) && !/(cannot|can't|unable|not able|no tool|don't have|not been sent)/i.test(output),
      hasQuote: /61,?234/.test(output),
    }
  } catch (e) { rec = { ...rec, error: String(e).slice(0, 300) } }
  appendFileSync(out!, JSON.stringify(rec) + "\n")
  console.log(JSON.stringify({ ...rec, output: undefined }))
}
